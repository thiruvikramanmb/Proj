#include <stdio.h>
#include <ctype.h>
#include <string.h>

int main()
{
    char str[100], word[30];
    int i = 0, j;

    printf("Enter a statement: ");
    fgets(str, sizeof(str), stdin);

    while (str[i] != '\0')
    {
        if (isspace(str[i]))
            i++;

        else if (isalpha(str[i]) || str[i] == '_')
        {
            j = 0;
            while (isalnum(str[i]) || str[i] == '_')
                word[j++] = str[i++];
            word[j] = '\0';

            if (!strcmp(word, "int") || !strcmp(word, "float") ||
                !strcmp(word, "char") || !strcmp(word, "if") ||
                !strcmp(word, "else") || !strcmp(word, "while") ||
                !strcmp(word, "return"))
                printf("%s - Keyword\n", word);
            else
                printf("%s - Identifier\n", word);
        }

        else if (isdigit(str[i]))
        {
            j = 0;
            while (isdigit(str[i]))
                word[j++] = str[i++];
            word[j] = '\0';

            printf("%s - Constant\n", word);
        }

        else if (strchr("+-*/%=<>", str[i]))
            printf("%c - Operator\n", str[i++]);

        else
            printf("%c - Special Symbol\n", str[i++]);
    }

    return 0;
}
