[0-9]+ {
    yylval.str = strdup(yytext);

type.y

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int yylex(void);
int yyerror(const char *s);

char* optimize(char *a, char op, char *b)
{
    char result[200];
    int x, y;

    int ca = (sscanf(a, "%d", &x) == 1);
    int cb = (sscanf(b, "%d", &y) == 1);

    /* Constant Folding */
    if (ca && cb) {
        switch(op) {
            case '+': sprintf(result, "%d", x+y); break;
            case '-': sprintf(result, "%d", x-y); break;
            case '*': sprintf(result, "%d", x*y); break;
            case '/':
                if(y == 0) {
                    printf("Error: Division by zero\n");
                    return strdup("ERROR");
                }
                sprintf(result, "%d", x/y);
                break;
            default:
                return strdup("ERROR");
        }

        printf("Constant Folding: %s %c %s = %s\n",
               a, op, b, result);

        return strdup(result);
    }

    /* Algebraic Transformation */
    if(op == '+') {
        if(strcmp(b,"0") == 0) {
            printf("Algebraic: x+0 -> x\n");
            return strdup(a);
        }
        if(strcmp(a,"0") == 0)
            return strdup(b);
    }

    if(op == '*') {
        if(strcmp(a,"0") == 0 || strcmp(b,"0") == 0) {
            printf("Algebraic: x*0 -> 0\n");
            return strdup("0");
        }
        if(strcmp(b,"1") == 0) {
            printf("Algebraic: x*1 -> x\n");
            return strdup(a);
        }
        if(strcmp(a,"1") == 0)
            return strdup(b);
    }

    if(op == '-') {
        if(strcmp(b,"0") == 0)
            return strdup(a);
    }

    if(op == '/') {
        if(strcmp(b,"1") == 0)
            return strdup(a);
    }

    /* Strength Reduction */
    if(op == '*' && cb && y == 2) {
        sprintf(result, "(%s+%s)", a, a);
        printf("Strength Reduction: x*2 -> x+x\n");
        return strdup(result);
    }

    if(op == '*' && cb && y == 4) {
        sprintf(result, "((%s+%s)+(%s+%s))",
                a, a, a, a);
        printf("Strength Reduction: x*4 -> (x+x)+(x+x)\n");
        return strdup(result);
    }

    sprintf(result, "(%s%c%s)", a, op, b);
    return strdup(result);
}
%}

%union {
    char *str;
}

%token <str> NUM ID
%type <str> expr

%left '+' '-'
%left '*' '/'
%right UMINUS

%%

program:
      /* empty */
    | program statement
    ;

statement:
      ID '=' expr ';'
      {
          printf("Optimized Expression: %s = %s\n\n",
                 $1, $3);
          free($1);
          free($3);
      }
    | expr ';'
      {
          printf("Optimized Expression: %s\n\n", $1);
          free($1);
      }
    ;

expr:
      NUM
      {
          $$ = $1;
      }
    | ID
      {
          $$ = $1;
      }
    | expr '+' expr
      {
          $$ = optimize($1, '+', $3);
          free($1);
          free($3);
      }
    | expr '-' expr
      {
          $$ = optimize($1, '-', $3);
          free($1);
          free($3);
      }
    | expr '*' expr
      {
          $$ = optimize($1, '*', $3);
          free($1);
          free($3);
      }
    | expr '/' expr
      {
          $$ = optimize($1, '/', $3);
          free($1);
          free($3);
      }
    | '-' expr %prec UMINUS
      {
          char temp[200];
          sprintf(temp, "(-%s)", $2);
          $$ = strdup(temp);
          free($2);
      }
    | '(' expr ')'
      {
          $$ = $2;
      }
    ;

%%

int yyerror(const char *s)
{
    printf("Syntax Error: %s\n", s);
    return 0;
}

int main()
{
    printf("Enter expressions (Ctrl+D to stop):\n\n");
    yyparse();
    return 0;
}


bison -dy type.y
flex type.l
gcc lex.yy.c y.tab.c -o optimize
./optimize
