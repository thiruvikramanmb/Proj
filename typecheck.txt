%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int yylex(void);
int yyerror(const char *s);
#define INT_TYPE 1
#define FLOAT_TYPE 2
#define ERROR_TYPE 0
struct Symbol {
char *name;
int type;
};
struct Symbol table[100];

int count = 0;
int lookup(const char *name)
{
int i;
for (i = 0; i < count; i++) {
if (strcmp(table[i].name, name) == 0)
return table[i].type;
}
return -1;
}
void declare(char *name, int type)
{
if (lookup(name) != -1) {
printf("Error: variable %s already declared\n", name);
free(name);
return;
}
if (count >= 100) {
printf("Error: symbol table is full\n");
free(name);
return;
}
table[count].name = name;
table[count].type = type;
count++;
printf("Declaration OK: %s\n", name);
}
int check_addition(int t1, int t2)
{
if (t1 == ERROR_TYPE || t2 == ERROR_TYPE)
return ERROR_TYPE;
if (t1 == t2)
return t1;
printf("Type error in expression: %s + %s\n",
t1 == INT_TYPE ? "int" : "float",
t2 == INT_TYPE ? "int" : "float");
return ERROR_TYPE;
}
%}
%union {
int type;
char *name;
}

%token INT FLOAT
%token <name> ID
%token INT_NUM FLOAT_NUM
%type <type> type expr
%left '+'
%%
program:
/* empty */
| program statement
;
statement:
declaration
| assignment
| error ';' { yyerrok; }
;
type:
INT { $$ = INT_TYPE; }
| FLOAT { $$ = FLOAT_TYPE; }
;
declaration:
type ID ';'
{
declare($2, $1);
}
;
assignment:
ID '=' expr ';'
{
int lhs = lookup($1);
if (lhs == -1) {
printf("Error: variable %s is undeclared\n", $1);
}
else if ($3 == ERROR_TYPE) {
printf("Assignment failed due to expression error\n");
}
else if (lhs != $3) {
printf("Type error: cannot assign %s to %s\n",
$3 == INT_TYPE ? "int" : "float",
lhs == INT_TYPE ? "int" : "float");
}
else {
printf("Assignment OK: %s = %s\n",
lhs == INT_TYPE ? "int" : "float",

$3 == INT_TYPE ? "int" : "float");
}
free($1);
}
;
expr:
ID
{
int t = lookup($1);
if (t == -1) {
printf("Error: variable %s is undeclared\n", $1);
$$ = ERROR_TYPE;
}
else {
$$ = t;
}
free($1);
}
| INT_NUM
{
$$ = INT_TYPE;
}
| FLOAT_NUM
{
$$ = FLOAT_TYPE;
}
| expr '+' expr
{
$$ = check_addition($1, $3);
}
;
%%
int yyerror(const char *s)
{
printf("Syntax error: %s\n", s);
return 0;
}
int main(void)
{
printf("Enter declarations and assignments (Ctrl+D to finish):\n");
yyparse();
return 0;

}
