
%{
#include <stdio.h>
#include <string.h>

char symbol[100][30];
int count = 0;

int search(char str[])
{
    for(int i = 0; i < count; i++)
        if(strcmp(symbol[i], str) == 0)
            return 1;

    return 0;
}

void insert(char str[])
{
    if(!search(str) && count < 100)
    {
        strcpy(symbol[count], str);
        count++;
    }
}
%}

%option noyywrap

letter [a-zA-Z_]
digit [0-9]

%%
"int"|"float"|"char"|"if"|"else"|"while"|"for"|"return" {
    printf("%s is a Keyword\n", yytext);
}

{letter}({letter}|{digit})* {
    printf("%s is an Identifier\n", yytext);
    insert(yytext);
}

{digit}+(\.{digit}+)? {
    printf("%s is a Number\n", yytext);
}

"+"|"-"|"*"|"/"|"="|">"|"<" {
    printf("%s is an Operator\n", yytext);
}

";"|","|"("|")"|"{"|"}" {
    printf("%s is a Special Symbol\n", yytext);
}

[ \t\n]+ ;
. ;
%%

int main()
{
    int i;

    printf("Enter the statement:\n");
    printf("Press Ctrl+D after typing.\n\n");

    yylex();

    printf("\nSYMBOL TABLE\n");
    printf("-----------------\n");

    for(i = 0; i < count; i++)
        printf("%d\t%s\n", i + 1, symbol[i]);

    return 0;
}
